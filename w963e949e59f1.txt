    try { body = await request.json(); } catch { return json({ error: 'bad request' }, 400, origin); }
    const lang = ['en', 'fr', 'es'].includes(body.lang) ? body.lang : 'en';
    let msgs = (Array.isArray(body.messages) ? body.messages : [])
      .filter(m => m && (m.role === 'user' || m.role === 'assistant') && typeof m.content === 'string' && m.content.trim())
      .map(m => ({ role: m.role, content: m.content.trim().slice(0, MAX_CHARS) }))
      .slice(-MAX_MSGS);
    while (msgs.length && msgs[0].role !== 'user') msgs.shift();
    // merge back-to-back turns from the same side
    msgs = msgs.reduce((a, m) => { if (a.length && a[a.length - 1].role === m.role) a[a.length - 1].content += '\n\n' + m.content; else a.push(m); return a; }, []);
    if (!msgs.length || msgs[msgs.length - 1].role !== 'user') return json({ error: 'bad request' }, 400, origin);

    // dry humour on roughly every second or third reply, not every one
    const turn = msgs.filter(m => m.role === 'user').length + (Array.isArray(body.messages) ? body.messages.filter(m => m && m.role === 'user').length - msgs.filter(m => m.role === 'user').length : 0);
    const aside = [1, 3].includes(turn % 5);
    const req = JSON.stringify({
      model: env.MODEL || 'claude-haiku-5-5',
      max_tokens: MAX_TOKENS,
      thinking: { type: 'disabled' }, // short chat answers: no hidden reasoning that could use up the reply
      system: [{ type: 'text', text: system(lang), cache_control: { type: 'ephemeral' } }, { type: 'text', text: aside ? 'NOTE: this reply MUST end with one line from the ASIDE BANK: choose the closest fit to the question (the general lines work for anything) and add it as the final sentence. Never reuse a line that already appears in an earlier reply in this conversation: pick an unused one, and if none fits, leave the aside out. The only exception: complaints, safety, payments or something going wrong get no aside.' : 'NOTE: no aside in this reply. No jokes, no quips: just answer, plainly and warmly.' }],
      messages: msgs,
    });
    const call = () => fetch('https://api.anthropic.com/v1/messages', {
      method: 'POST',
      headers: { 'x-api-key': env.ANTHROPIC_API_KEY, 'anthropic-version': '2023-06-01', 'content-type': 'application/json' },
      body: req,
    });
    // one quick retry when the AI service is busy or has a hiccup
    let r = await call();
    if (r.status === 429 || r.status >= 500) { await new Promise(z => setTimeout(z, 700)); r = await call(); }
    if (!r.ok) { console.log('ai error', r.status, (await r.text()).slice(0, 400)); return json({ error: 'ai ' + r.status }, 502, origin); }
    const out = await r.json();
    let text = (out.content || []).filter(c => c.type === 'text').map(c => c.text).join('').trim();

    const actions = [];
    text = text.replace(/\[([A-Z_]+)\]/g, (m, k) => {
      if (ACTS[k] && actions.length < 2 && !actions.some(a => a[1] === ACTS[k][1])) actions.push([ACTS[k][0][lang], ACTS[k][1]]);
      return '';
    }).replace(/\*\*|__|^#+\s*/gm, '').replace(/[ \t]+\n/g, '\n').trim();
    // safety net: contact details live on the buttons below the chat
    text = text.replace(/\s*(?:at|au|al|:)?\s*\+?1?\s*\(?877\)?[\s.-]*395[\s.-]*7435/gi, '').replace(/\s*(?:at|à|a)?\s*[a-z]+@americangraphiccontractors\.com/gi, '').replace(/\s+(?:le|al|at|to|à|a)(?=\s*[,.;:!?])/gi, '').replace(/\s+([,.;:!?])/g, '$1').replace(/\(\s*\)/g, '').trim();
    // never say the same sentence twice
    { const seen = new Set(); text = text.split(/(?<=[.!?…])\s+/).filter(x => { const k = x.toLowerCase().replace(/[^a-zà-ÿ0-9]+/g, ''); if (!k || seen.has(k)) return false; seen.add(k); for (const p of seen) if (p !== k && (p.includes(k) || k.includes(p)) && k.length > 30) return false; return true; }).join(' '); }
    // keep replies short: if the model runs long, keep whole sentences up to about 55 words
    const sents = text.split(/(?<=[.!?…])\s+/);
    if (text.split(/\s+/).length > 60 && sents.length > 1) {
      // keep the opening answer and the closing dry aside; drop middle sentences first
      const last = sents[sents.length - 1], lw = last.split(/\s+/).length, tail = lw <= 16 ? [last] : [];
      const body = tail.length ? sents.slice(0, -1) : sents;
      let keep = [], n = tail.length ? lw : 0;
      for (const x of body) { const w = x.split(/\s+/).length; if (keep.length && n + w > 58) break; keep.push(x); n += w; }
      text = keep.concat(tail).join(' ');
    }
    if (!text) { console.log('empty reply', JSON.stringify(out).slice(0, 400)); return json({ error: 'empty' }, 502, origin); }
    return json({ reply: text, actions }, 200, origin);
  },
